const { chromium } = require('playwright');
const fs = require('fs'), path = require('path');
const [dir, out] = process.argv.slice(2);
(async () => {
  fs.mkdirSync(out, { recursive: true });
  const b = await chromium.launch();
  for (const f of fs.readdirSync(dir).filter((x) => x.endsWith('.html')).sort()) {
    const p = await b.newPage({ viewport: { width: 1700, height: 1200 }, deviceScaleFactor: 2 });
    const errs = []; p.on('pageerror', (e) => errs.push(String(e))); p.on('requestfailed', (r) => errs.push('failed ' + r.url().slice(0, 80)));
    await p.goto('file://' + path.join(dir, f)); await p.evaluate(() => document.fonts.ready); await p.waitForTimeout(300);
    const el = await p.$('.sheet');
    // overflow check: any element sticking out of the sheet
    const over = await p.evaluate(() => { const s = document.querySelector('.sheet').getBoundingClientRect(); return [...document.querySelectorAll('.sheet *')].filter((e) => { const r = e.getBoundingClientRect(); return r.width && (r.right > s.right + 1 || r.bottom > s.bottom + 1); }).slice(0, 5).map((e) => e.className || e.tagName); });
    await el.screenshot({ path: path.join(out, f.replace('.html', '.png')) });
    console.log(f, errs.length ? errs : 'no errors', over.length ? 'OVERFLOW ' + over : '');
    await p.close();
  }
  await b.close();
})();
